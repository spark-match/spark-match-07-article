\begin{lstlisting}[style=pythonstyle, caption={Definicion del agente LangChain}, label={lst:agent}]
from langchain.agents import create_tool_calling_agent, AgentExecutor
from langchain_aws import ChatBedrock

llm = ChatBedrock(
    model_id="anthropic.claude-3-haiku-20240307-v1:0",
    region_name="us-east-1",
    model_kwargs={"temperature": 0.2, "max_tokens": 2048},
)

tools = [extract_preferences, query_careers, score_rank, justify_results]
agent = create_tool_calling_agent(llm, tools, prompt)
executor = AgentExecutor(agent=agent, tools=tools, verbose=True, max_iterations=5)
\end{lstlisting}
